//! One-sample runs test over k categories (spec
//! docs/superpowers/specs/2026-10-13-runs-test-design.md).
//!
//! R = the number of maximal runs of equal consecutive values — the length of an Arrow
//! run-end-encoded array's run_ends. Under random ordering of the same multiset (n rows,
//! category counts nᵢ, S₂ = Σnᵢ², S₃ = Σnᵢ³; null is a category), Barton & David (1957):
//!
//!   E[R]   = 1 + (n² − S₂) / n
//!   Var[R] = [S₂(S₂ + n(n+1)) − 2nS₃ − n³] / (n²(n−1))
//!
//! z = (R − E[R]) / √Var[R]; p = Φ(z), ONE-SIDED LOWER TAIL: P(R ≤ observed), "fewer runs
//! than chance" — the only direction relevant to REE. Normal approximation, no continuity
//! correction. Var = 0 (n = 1, one distinct value, every value distinct) → z = p = NaN.
//!
//! One pass per column, hashing once per RUN, not per row: each 2^16-row chunk is scanned
//! comparing neighbours (null-aware; floats by canonical bits), adding each run's length to
//! its value's count. Chunk partials merge in row order (rayon's `reduce` keeps operand
//! order): runs add, minus one when the left chunk's last value equals the right's first.
//! Keys are compared, not only hashed, so results are exact.

use std::collections::HashMap;
use std::hash::Hash;

use foldhash::fast::RandomState;
use polars::prelude::*;
use polars_arrow::array::Array;
use rayon::prelude::*;
use statrs::distribution::{ContinuousCDF, Normal};

use crate::common::encode::{canon_f32, canon_f64};

const CHUNK: usize = 1 << 16;
const ONE_CHUNK: &str = "rechunked to one chunk";

// ─────────────────────────────────────────────────────────────────────────────
// Runs over row ranges
// ─────────────────────────────────────────────────────────────────────────────

/// Summary of one contiguous range of rows. Keys: `None` = null. `first` / `last` are
/// `None` only for an empty range.
#[derive(Debug, Clone, PartialEq)]
struct Partial<K: Eq + Hash> {
    rows: u64,
    runs: u64,
    first: Option<Option<K>>,
    last: Option<Option<K>>,
    counts: HashMap<Option<K>, u64, RandomState>,
}

impl<K: Copy + Eq + Hash> Partial<K> {
    fn empty() -> Self {
        Self {
            rows: 0,
            runs: 0,
            first: None,
            last: None,
            counts: HashMap::default(),
        }
    }

    /// One pass over `values` (in row order): one hash-map update per run.
    fn scan(mut values: impl Iterator<Item = Option<K>>) -> Self {
        let mut p = Self::empty();
        let Some(mut cur) = values.next() else {
            return p;
        };
        p.first = Some(cur);
        let mut len = 1u64;
        for v in values {
            if v == cur {
                len += 1;
            } else {
                p.close(cur, len);
                cur = v;
                len = 1;
            }
        }
        p.close(cur, len);
        p.last = Some(cur);
        p
    }

    fn close(&mut self, value: Option<K>, len: u64) {
        *self.counts.entry(value).or_insert(0) += len;
        self.runs += 1;
        self.rows += len;
    }

    /// `self` followed by `other` (row order matters; associative). A run spanning the
    /// boundary was counted once on each side, so one is taken off.
    fn merge(mut self, mut other: Self) -> Self {
        if self.rows == 0 {
            return other;
        }
        if other.rows == 0 {
            return self;
        }
        let joined = self.last == other.first;
        if self.counts.len() < other.counts.len() {
            std::mem::swap(&mut self.counts, &mut other.counts);
        }
        for (k, c) in other.counts {
            *self.counts.entry(k).or_insert(0) += c;
        }
        Self {
            rows: self.rows + other.rows,
            runs: self.runs + other.runs - joined as u64,
            first: self.first,
            last: other.last,
            counts: self.counts,
        }
    }

    fn tally(&self) -> Tally {
        let (mut s2, mut s3) = (0u128, 0u128);
        for &c in self.counts.values() {
            let c = c as u128;
            s2 += c * c;
            s3 += c * c * c; // fits u128 for n up to ≈ 7·10¹²
        }
        Tally {
            rows: self.rows,
            runs: self.runs,
            s2,
            s3,
        }
    }
}

/// What the statistics need from a column: n, R, S₂ = Σnᵢ², S₃ = Σnᵢ³.
#[derive(Debug, Clone, Copy, Default, PartialEq)]
struct Tally {
    rows: u64,
    runs: u64,
    s2: u128,
    s3: u128,
}

/// Rows `0..n` in parallel 2^16-row chunks, merged in row order; `get(i)` is row i's key.
fn runs_of<K, G>(n: usize, get: G) -> Tally
where
    K: Copy + Eq + Hash + Send,
    G: Fn(usize) -> Option<K> + Sync,
{
    (0..n.div_ceil(CHUNK))
        .into_par_iter()
        .map(|c| Partial::scan((c * CHUNK..((c + 1) * CHUNK).min(n)).map(&get)))
        .reduce(Partial::empty, Partial::merge)
        .tally()
}

// ─────────────────────────────────────────────────────────────────────────────
// Statistics
// ─────────────────────────────────────────────────────────────────────────────

/// One column's result row.
#[derive(Debug, Clone, Copy, PartialEq)]
pub(crate) struct RunsStats {
    pub n_rows: u64,
    pub n_runs: u64,
    pub expected_runs: f64,
    pub z: f64,
    pub p_value: f64,
    pub mean_run_length: f64,
}

/// S₂(S₂ + n(n+1)) − 2nS₃ − n³: exact in i128 — its terms (~n⁴) cancel heavily on
/// near-constant columns, which f64 would lose — and f64 once i128 overflows
/// (n ≳ 3·10⁹).
fn variance_numerator(n: u128, s2: u128, s3: u128) -> f64 {
    let exact = || -> Option<i128> {
        let (n, s2, s3) = (
            i128::try_from(n).ok()?,
            i128::try_from(s2).ok()?,
            i128::try_from(s3).ok()?,
        );
        let a = s2.checked_mul(s2.checked_add(n.checked_mul(n + 1)?)?)?;
        let b = n.checked_mul(2)?.checked_mul(s3)?;
        let c = n.checked_mul(n)?.checked_mul(n)?;
        a.checked_sub(b)?.checked_sub(c)
    };
    exact().map_or_else(
        || {
            let (n, s2, s3) = (n as f64, s2 as f64, s3 as f64);
            s2 * (s2 + n * (n + 1.0)) - 2.0 * n * s3 - n * n * n
        },
        |v| v as f64,
    )
}

/// 0/0 gives NaN naturally: n = 0 (everything), n = 1 and Var = 0 (z, p).
fn statistics(t: Tally) -> RunsStats {
    let n = t.rows as u128;
    let (nf, rf) = (t.rows as f64, t.runs as f64);
    let expected = 1.0 + (n * n - t.s2) as f64 / nf;
    let variance = variance_numerator(n, t.s2, t.s3) / ((n * n) as f64 * (nf - 1.0));
    let z = (rf - expected) / variance.sqrt();
    let p_value = if z.is_nan() {
        f64::NAN
    } else {
        Normal::new(0.0, 1.0).expect("standard normal").cdf(z)
    };
    RunsStats {
        n_rows: t.rows,
        n_runs: t.runs,
        expected_runs: expected,
        z,
        p_value,
        mean_run_length: nf / rf,
    }
}

fn supported(dtype: &DataType) -> bool {
    matches!(
        dtype,
        DataType::Int8
            | DataType::Int16

            | DataType::Int64
            | DataType::UInt8
            | DataType::UInt16

            | DataType::Float32

            | DataType::Duration(_)

            | DataType::Binary

            | DataType::Enum(_, _)

/// Integer keys (temporal values, Decimal's unscaled i128, Categorical / Enum codes).
fn numeric<T>(ca: &ChunkedArray<T>) -> Tally
where
    T: PolarsNumericType,
    T::Native: Eq + Hash,
{
    let arr = ca.downcast_iter().next().expect(ONE_CHUNK);
    runs_of(arr.len(), |i| arr.is_valid(i).then(|| arr.value(i)))
}

/// The run statistics of one column, in its row order. Keys by physical type:
/// Categorical / Enum compare codes (bijective with their strings within a column),
/// Decimal its unscaled i128, temporal types their integers, floats canonical bits.
pub(crate) fn series_runs(s: &Series) -> PolarsResult<RunsStats> {
    if !supported(s.dtype()) {
        polars_bail!(
            InvalidOperation: "runs_test: unsupported dtype {} (column {})", s.dtype(), s.name()
        );
    }
    if s.is_empty() {
        return Ok(statistics(Tally::default()));
    }
    // One chunk, so row i is index i of a single Arrow array (zero-copy if already one).
    let phys = s.to_physical_repr().rechunk();

        DataType::Int8 => numeric(phys.i8()?),
        DataType::Int16 => numeric(phys.i16()?),
        DataType::Int32 => numeric(phys.i32()?),
        DataType::Int64 => numeric(phys.i64()?),
        DataType::UInt8 => numeric(phys.u8()?),
        DataType::UInt16 => numeric(phys.u16()?),
        DataType::UInt32 => numeric(phys.u32()?),
        DataType::UInt64 => numeric(phys.u64()?),
        DataType::Int128 => numeric(phys.i128()?),
        DataType::Float32 => {
            let arr = phys.f32()?.downcast_iter().next().expect(ONE_CHUNK);
            runs_of(arr.len(), |i| arr.is_valid(i).then(|| canon_f32(arr.value(i))))
        }
        DataType::Float64 => {
            let arr = phys.f64()?.downcast_iter().next().expect(ONE_CHUNK);
            runs_of(arr.len(), |i| arr.is_valid(i).then(|| canon_f64(arr.value(i))))
        }
        DataType::Boolean => {
            let arr = phys.bool()?.downcast_iter().next().expect(ONE_CHUNK);
            runs_of(arr.len(), |i| arr.is_valid(i).then(|| arr.value(i)))
        }
        DataType::String => {
            let arr = phys.str()?.downcast_iter().next().expect(ONE_CHUNK);
            runs_of(arr.len(), |i| arr.is_valid(i).then(|| arr.value(i)))
        }

            let arr = phys.binary()?.downcast_iter().next().expect(ONE_CHUNK);

        }
        dt => polars_bail!(ComputeError: "runs_test: unexpected physical dtype {dt}"),
    };
    Ok(statistics(tally))
}

def runs_int(
    n_rows: int,
    n_cols: int,
    mean_run: int,
    cardinality: int,
    prefix: str = "c",
    seed: int = 42,
) -> pl.DataFrame:
    """Int64 columns {prefix}0… made of runs: geometric run lengths with mean `mean_run`
    (1: every row its own draw), each run's value uniform in [0, cardinality). Adjacent
    runs drawing the same value merge (probability 1 / cardinality)."""
    rng = np.random.default_rng(seed)
    columns = {}
    for i in range(n_cols):
        lengths = rng.geometric(1 / mean_run, size=n_rows // mean_run + 64)
        while lengths.sum() < n_rows:
            more = rng.geometric(1 / mean_run, size=n_rows // mean_run + 64)
            lengths = np.concatenate([lengths, more])
        values = rng.integers(0, cardinality, lengths.size, dtype=np.int64)
        columns[f"{prefix}{i}"] = np.repeat(values, lengths)[:n_rows]
    return pl.DataFrame(columns)


def runs_str(
    n_rows: int,
    n_cols: int,
    mean_run: int,
    cardinality: int,
    prefix: str = "s",
    seed: int = 42,
) -> pl.DataFrame:
    """runs_int's runs as String values "v<k>"."""
    df = runs_int(n_rows, n_cols, mean_run, cardinality, prefix, seed)
    return df.with_columns(("v" + pl.col(c).cast(pl.String)).alias(c) for c in df.columns)

"""One-sample runs test over k categories: does row order cluster equal values? (spec
docs/superpowers/specs/2026-10-13-runs-test-design.md)"""

import polars as pl

from analytics._dtypes import INTEGERS_64
from analytics.base import Technique

# Mirrors the Rust kernel's `supported` (src/techniques/single_column/ordered/runs_test.rs).
ELIGIBLE = (
    *INTEGERS_64,
    pl.Boolean,
    pl.Float32,
    pl.Float64,
    pl.Decimal,
    pl.Date,
    pl.Datetime,
    pl.Duration,
    pl.Time,
    pl.String,
    pl.Binary,
    pl.Categorical,
    pl.Enum,
)


class RunsTest(Technique):
    """Runs of equal consecutive values per column, and whether row order makes them
    fewer than chance would — the signal for Arrow run-end encoding (REE).

    `n_runs` (R) is the number of maximal runs of equal consecutive values: exactly the
    length of an REE array's run_ends. Nulls are one value (consecutive nulls are one
    run, as in REE); floats compare canonically (−0.0 = 0.0, every NaN equal).

    Under random ordering of the same values — n rows, k categories of counts nᵢ; the
    k-category generalisation of the one-sample Wald–Wolfowitz runs test (Barton & David
    1957):

        E[R]   = 1 + (n² − Σnᵢ²) / n
        Var[R] = [Σnᵢ²(Σnᵢ² + n(n+1)) − 2nΣnᵢ³ − n³] / (n²(n−1))

    `z` = (R − E[R]) / √Var[R]. `p_value` is ONE-SIDED, LOWER TAIL: Φ(z) = P(R ≤ observed),
    "fewer runs than chance" — the only direction that matters for REE. Columns that
    alternate more than chance (A B A B …) get z > 0 and p near 1. p uses the asymptotic
    normal approximation, without continuity correction. Var = 0 (one row, one distinct
    value, or every value distinct: R is then fixed) → z and p are NaN.

    p is not the effect size: at large n it vanishes even when runs are barely longer than
    one row. `mean_run_length` = n / R is the effect size; p says only whether the
    clustering is real. The technique reports metrics only — whether REE pays off for a
    dtype is the recommenders' decision.

    Order-dependent: implementations read rows in the frame's own order. Eligible:
    integers (≤ 64-bit), Boolean, Float, Decimal, temporal, String, Binary, Categorical and
    Enum columns with at least one row; List / Array / Struct / Null / Object, 128-bit
    integers and zero-row columns are ineligible. `dtype` (Python's str(dtype)) is
    reported on every row.
    """

    SCOPE = "per_column"
    ARITY = 1
    DESCRIPTORS = {"dtype": pl.String}
    METRICS = {
        "n_rows": pl.UInt64,
        "n_runs": pl.UInt64,
        "expected_runs": pl.Float64,
        "z": pl.Float64,
        "p_value": pl.Float64,
        "mean_run_length": pl.Float64,
    }
    RTOL = 1e-9
    # p underflows to 0 near z ≈ −38.5; two erfc implementations may reach the subnormal
    # range one value apart.
    ATOL = 1e-300

    def eligible(self, series: pl.Series) -> bool:
        return isinstance(series.dtype, ELIGIBLE) and series.len() > 0

    def describe(self, frames, combos):
        dtypes = {
            (n, c): str(dt) for n, f in frames.items() for c, dt in f.schema.items()
        }
        return {"dtype": [dtypes[n, c] for ((n, c),) in combos]}

```python
import math

import polars as pl

from analytics.runs_test.base import RunsTest


def exact_moments(n: int, counts: list[int]) -> tuple[float, float]:
    """(E[R], Var[R]) under random ordering of n rows with category `counts`, in exact
    integer arithmetic up to the final divisions (Var NaN when n < 2)."""
    s2 = sum(c * c for c in counts)
    s3 = sum(c**3 for c in counts)
    expected = 1 + (n * n - s2) / n
    numerator = s2 * (s2 + n * (n + 1)) - 2 * n * s3 - n**3
    denominator = n * n * (n - 1)
    return expected, numerator / denominator if denominator else math.nan


def _canonical(s: pl.Series) -> pl.Series:
    """−0.0 → 0.0 and every NaN payload → one NaN, as the Rust kernel compares floats
    (group_by would otherwise keep −0.0 and 0.0 apart)."""
    if s.dtype in (pl.Float32, pl.Float64):
        return (s + 0.0).fill_nan(float("nan"))
    return s


def runs_statistics(s: pl.Series) -> dict[str, int | float]:
    """The RunsTest metrics of one non-empty column."""
    s = _canonical(s)
    n = s.len()
    after, before = s.slice(1), s.slice(0, n - 1)
    changed = after.ne_missing(before)
    if s.dtype in (pl.Float32, pl.Float64):  # NaN continues a NaN run, whatever `!=` says
        changed = changed & ~(after.is_nan() & before.is_nan()).fill_null(False)
    n_runs = 1 + int(changed.sum())
    counts = [int(c) for c in pl.DataFrame({"v": s}).group_by("v").len()["len"]]
    expected, variance = exact_moments(n, counts)
    z = (n_runs - expected) / math.sqrt(variance) if variance > 0 else math.nan
    p = math.nan if math.isnan(z) else 0.5 * math.erfc(-z / math.sqrt(2))
    return {
        "n_rows": n,
        "n_runs": n_runs,
        "expected_runs": expected,
        "z": z,
        "p_value": p,
        "mean_run_length": n / n_runs,
    }


class RunsTestPolars(RunsTest):
    """Reference: R from neighbour comparisons (`ne_missing`, null-aware), category
    counts from a group-by (nulls one group), moments in exact Python ints; only the final
    divisions and Φ (math.erfc) are floating point. Shares no code with the Rust kernel."""

    def _compute(self, frames, combos):
        rows = [runs_statistics(frames[f][c]) for ((f, c),) in combos]
        return self.metrics_frame(
            combos, {m: [r[m] for r in rows] for m in self.METRICS}
        )
```

# 4. Statistical correctness (no conclusions to test: the technique reports metrics only)


def test_moments_match_permutation_monte_carlo():
    counts, n = [16, 12, 8, 4], 40
    base = np.repeat(np.arange(len(counts)), counts)
    shuffles = np.random.default_rng(7).permuted(np.tile(base, (20_000, 1)), axis=1)
    runs = 1 + (shuffles[:, 1:] != shuffles[:, :-1]).sum(axis=1)
    expected, variance = exact_moments(n, counts)
    assert abs(runs.mean() - expected) < 4 * math.sqrt(variance / runs.size)
    assert runs.var(ddof=1) == pytest.approx(variance, rel=0.05)


@pytest.mark.parametrize("impl", ALL)
@pytest.mark.parametrize("seed", range(5))

    sm_runs = pytest.importorskip(

@pytest.mark.parametrize("impl", ALL)
def test_near_constant_variance_is_exact(impl):
    # Var's numerator terms are ~1e24 here and cancel to ~4e14: f64 would lose ~7 digits.
    n = 1_000_000
    x = np.zeros(n, dtype=np.int64)
    x[::100_000] = 1
    n_runs = 1 + int(np.count_nonzero(np.diff(x)))
    expected, variance = exact_moments(n, [n - 10, 10])
    r = stats(impl, pl.Series("x", x))
    assert r["n_runs"] == n_runs
    assert r["z"] == pytest.approx((n_runs - expected) / math.sqrt(variance), rel=1e-9)
